花瓣飘落 petals.js

单文件、零依赖，引入即自动启用。当前页面的花瓣就由它渲染。

实时参数

状态：运行中

说明

  • 引入即自动启用，无需调用初始化
  • 贴图 base64 内联，无额外图片请求
  • 标签页隐藏时自动暂停，后台不耗电
  • SPA 切换路由前调用 Petals.destroy()

接入代码

<!-- 1. 最简：引入即生效（放 head 或 body 末尾都可以） -->
<script src="https://cdn.jsdelivr.net/gh/bufork/cdn-assets@main/effects/petals/petals.js"></script>
<!-- 2. 自定义参数：配置脚本必须写在引入之前 -->
<!-- 下面为示例值，可自行修改；不写则使用默认值（等同上方滑块初始值） -->
<script>
window.PETALS_CONFIG = {
    count: 30,     // 花瓣数量
    size: 1,       // 尺寸倍率
    speed: 0.8,    // 下落速度
    opacity: 0.9,  // 不透明度 0–1
    sway: true,    // 左右摇摆
    zIndex: 9999   // 图层层级
};
</script>
<script src="https://cdn.jsdelivr.net/gh/bufork/cdn-assets@main/effects/petals/petals.js"></script>
<!-- 3. 运行时控制 -->
Petals.set({ count: 20, speed: 1.5 });  // 改参数，立即生效
Petals.disable();                       // 暂停
Petals.enable();                        // 恢复
Petals.destroy();                       // 移除画布与监听
<!-- 4. 只加载不自动播放（例如等用户点开关） -->
<script>window.PETALS_AUTO = false</script>
<script src="https://cdn.jsdelivr.net/gh/bufork/cdn-assets@main/effects/petals/petals.js"></script>
<script>Petals.enable()</script>
